/**

 * Â«Probar sesiÃ³nÂ» deja la fuente ACTIVA (sesiÃ³n cifrada hasta sesion_valida_hasta),
 * REQUIERE_VERIFICACION (la tienda pidiÃ³ cÃ³digo o CAPTCHA) o CON_ERROR. Si al probar faltan
 * las reglas de ingreso, queda SIN_PROBAR; al renovar (D-41) queda CON_ERROR. Antes de buscar,
 * {@link #sesionVigente} renueva sola la sesiÃ³n vencida; si no puede, la tienda queda fuera
 * de la bÃºsqueda sin romperla.
 */
@Service
@RequiredArgsConstructor
public class SesionFuenteService {

    /** Una sesiÃ³n que vence en menos de esto se renueva antes de buscar. */
    static final Duration MARGEN_RENOVACION = Duration.ofMinutes(10);
    private static final String PROCESO = "SESION_FUENTE";
    private static final int LARGO_ERROR = 500;

    private final FuenteBusquedaConfigRepository fuenteRepository;
    private final FuenteCredencialRepository credencialRepository;
    private final CredencialesFuenteService credenciales;
    private final FuenteSelectorRepository selectorRepository;
    private final FuentesConfigService config;
    private final LectorSesion lector;
    private final UsuarioRepository usuarioRepository;

    @Transactional
    public FuenteConfigResponse probar(String codigoTexto) {
        CodigoFuente codigo = FuentesConfigService.codigo(codigoTexto);
        FuenteBusquedaConfig fuente = fuente(codigo);
        if (!credenciales.tiene(codigo, TipoCredencial.USUARIO) || !credenciales.tiene(codigo, TipoCredencial.CLAVE)) {
            throw new BusinessException("Primero guarda la cuenta de bÃºsqueda (usuario y clave) de " + fuente.getNombre());
        }
        iniciar(fuente, true);
        return config.aRespuesta(fuente);
    }

    /** La sesiÃ³n en claro para buscar en la tienda, renovada si venciÃ³. VacÃ­a si la tienda no puede entrar. */
    @Transactional
    public Optional<String> sesionVigente(CodigoFuente codigo) {
        FuenteBusquedaConfig fuente = fuente(codigo);
        if (fuente.getEstado() != EstadoFuente.ACTIVA) {
            return Optional.empty();
        }
        if (!configuracionCompleta(codigo)) {
            aplicar(fuente, false, ResultadoSesion.sinConfigurar("Fuente omitida: sin configurar."));
            return Optional.empty();
        }
        LocalDateTime hasta = fuente.getSesionValidaHasta();
        if (hasta != null && hasta.isAfter(LocalDateTime.now().plus(MARGEN_RENOVACION))) {
            Optional<String> guardada = credenciales.leer(codigo, TipoCredencial.SESION);
            if (guardada.isPresent()) {
                return guardada;
            }
        }
        return iniciar(fuente, false) ? credenciales.leer(codigo, TipoCredencial.SESION) : Optional.empty();
    }

    /**
     * 9-B.1 (tarea F): la tienda pidiÃ³ cÃ³digo o CAPTCHA mientras se leÃ­a. La fuente queda
     * REQUIERE_VERIFICACION, sin sesiÃ³n y fuera de la bÃºsqueda hasta que se vuelva a probar (9-A).
     */
    @Transactional
    public void marcarVerificacion(CodigoFuente codigo, String motivo) {
        aplicar(fuente(codigo), false, ResultadoSesion.verificacion(
                motivo != null && !motivo.isBlank() ? motivo : "La tienda pidiÃ³ un cÃ³digo o CAPTCHA."));
    }

    /** Comprueba cuenta y reglas mÃ­nimas antes de cuota, pausa o renovaciÃ³n de sesiÃ³n. */
    public boolean configuracionCompleta(CodigoFuente codigo) {
        return credenciales.tiene(codigo, TipoCredencial.USUARIO)
                && credenciales.tiene(codigo, TipoCredencial.CLAVE)
                && ReglasMinimas.faltantes(
                        selectorRepository.findByFuente_CodigoAndStatus(codigo, "A")).isEmpty();
    }

    /**
     * Pide al lector iniciar sesiÃ³n y deja la fuente en el estado que corresponde. true si quedÃ³ ACTIVA.
     * manual: Â«Probar sesiÃ³nÂ» del Jefe (9-A). false: renovaciÃ³n antes de buscar (D-41).
     */
    private boolean iniciar(FuenteBusquedaConfig fuente, boolean manual) {
        CodigoFuente codigo = fuente.getCodigo();
        String usuarioTienda = credenciales.leer(codigo, TipoCredencial.USUARIO).orElse(null);
        String claveTienda = credenciales.leer(codigo, TipoCredencial.CLAVE).orElse(null);
        ResultadoSesion r = (usuarioTienda == null || claveTienda == null)
                ? ResultadoSesion.sinConfigurar("Falta la cuenta de bÃºsqueda de " + fuente.getNombre())
                : lector.iniciarSesion(codigo, fuente.getUrlBase(), usuarioTienda, claveTienda);
        if (r == null) {
            r = ResultadoSesion.error("El servicio de lectura no respondiÃ³.");
        }
        return aplicar(fuente, manual, r);
    }

    /** Deja la fuente en el estado que corresponde al resultado. true si quedÃ³ ACTIVA. */
    private boolean aplicar(FuenteBusquedaConfig fuente, boolean manual, ResultadoSesion r) {
        CodigoFuente codigo = fuente.getCodigo();
        String nombre = usuarioActual();
        Long idUsuario = idUsuario(nombre);
        boolean ok = r.resultado() == ResultadoSesion.Resultado.OK && r.sesion() != null && r.validaHasta() != null;
        if (ok) {
            credenciales.guardar(codigo, TipoCredencial.SESION, r.sesion(), idUsuario, nombre);
            fuente.setEstado(EstadoFuente.ACTIVA);
            fuente.setSesionValidaHasta(r.validaHasta());
            fuente.setUltimoError(null);
        } else {
            credencialRepository.findByFuente_CodigoAndTipo(codigo, TipoCredencial.SESION)
                    .ifPresent(credencialRepository::delete);
            fuente.setEstado(estadoSiNoOk(r, manual));
            fuente.setSesionValidaHasta(null);
            fuente.setUltimoError(recortar(r.mensaje() != null ? r.mensaje() : "No se pudo iniciar sesiÃ³n en la tienda."));
        }
        fuente.setUltimaPrueba(LocalDateTime.now());
        fuente.setEditadoPor(idUsuario);
        fuente.setUserUpdate(nombre);
        fuente.setProcessUpdate(PROCESO);
        fuente.setDateUpdate(LocalDateTime.now());
        fuenteRepository.save(fuente);
        return ok;
    }

    /**
     * Un fallo de conexiÃ³n, un tiempo agotado o un bloqueo quedan en CON_ERROR.
     * En Â«Probar sesiÃ³nÂ», faltar la cuenta o las reglas de ingreso no es un fallo de la tienda:
     * SIN_PROBAR. TambiÃ©n en la renovaciÃ³n antes de buscar (D-41): falta de configuraciÃ³n
     * reemplaza la decisiÃ³n anterior de CON_ERROR y no genera aviso al Jefe.

    private static EstadoFuente estadoSiNoOk(ResultadoSesion r, boolean manual) {
        return switch (r.resultado()) {
            case VERIFICACION -> EstadoFuente.REQUIERE_VERIFICACION;
            case SIN_CONFIGURAR -> EstadoFuente.SIN_PROBAR;
            default -> EstadoFuente.CON_ERROR;
        };
    }

    private FuenteBusquedaConfig fuente(CodigoFuente codigo) {
        return fuenteRepository.findByCodigo(codigo)
                .orElseThrow(() -> new ResourceNotFoundException("No existe la fuente " + codigo));
    }

    private static String recortar(String texto) {
        return texto.length() <= LARGO_ERROR ? texto : texto.substring(0, LARGO_ERROR);
    }

    private static String usuarioActual() {
        var auth = SecurityContextHolder.getContext().getAuthentication();
        return auth != null ? auth.getName() : "SISTEMA";
    }

    private Long idUsuario(String nombre) {
        return usuarioRepository.findByNombreUsuario(nombre).map(u -> u.getIdUsuario()).orElse(null);
    }
}

public final class PuenteLector {

    /** Si la sesiÃ³n guardada no trae su vencimiento, se manda con este margen (SesionFuenteService ya la validÃ³). */
    static final Duration VIGENCIA_SIN_FECHA = Duration.ofMinutes(30);
    private static final int LARGO_CONSULTA = 200;

    private static final ObjectMapper JSON = new ObjectMapper();

    private PuenteLector() {
    }

    /**
     * La sesiÃ³n que manda el despachador, en el formato del servicio de lectura.
     * Acepta {"validaHasta": "...", "storageState": {...}} o el storageState solo
     * ({"cookies": [...], "origins": [...]}). VacÃ­a â†’ null (se lee sin sesiÃ³n).
     *
     * @throws IllegalArgumentException si no es ninguno de los dos formatos
     */
    public static ContratoLector.Sesion sesion(String sesion, Clock reloj) {
        if (sesion == null || sesion.isBlank()) {
            return null;
        }
        JsonNode nodo;
        try {
            nodo = JSON.readTree(sesion);
        } catch (Exception e) {
            throw new IllegalArgumentException("La sesiÃ³n no es JSON.");
        }
        if (nodo == null || !nodo.isObject()) {
            throw new IllegalArgumentException("La sesiÃ³n no es un objeto JSON.");
        }
        String porDefecto = Instant.now(reloj).plus(VIGENCIA_SIN_FECHA).toString();
        if (nodo.has("storageState")) {
            return new ContratoLector.Sesion(instante(nodo.path("validaHasta").asText(null), porDefecto),
                    nodo.get("storageState"));
        }
        if (nodo.has("cookies")) {
            return new ContratoLector.Sesion(porDefecto, nodo);
        }
        throw new IllegalArgumentException("La sesiÃ³n no trae cookies.");
    }

    /** Las fichas que leyÃ³ el servicio de lectura. Sin fichas â†’ SIN_RESULTADOS. */
    public static ResultadoLectura aResultado(ContratoLector.Respuesta respuesta) {
        if (respuesta == null || respuesta.resultados() == null) {
            return ResultadoLectura.sinResultados();
        }
        List<FichaFuente> fichas = new ArrayList<>();
        for (ContratoLector.Resultado r : respuesta.resultados()) {
            if (r != null) {
                fichas.add(aFicha(r));
            }
        }
        return ResultadoLectura.ok(fichas);
    }

    /** El precio va en texto plano ("1234.50") y la moneda en ISO, como los espera NormalizadorFicha. */
    static FichaFuente aFicha(ContratoLector.Resultado r) {

    public static ResultadoLectura deError(String tienda, LectorRemotoException e) {

    /** QuÃ© se escribe en el buscador; null si no hay nada que buscar (o es bÃºsqueda por imagen). */
    public static String textoDeBusqueda(ConsultaBusqueda consulta) {
        if (consulta == null || consulta.metodo() == null) {
            return null;
        }
        String texto = switch (consulta.metodo()) {
            case NUMERO_PARTE -> unir(consulta.marca(), consulta.numeroParte());
            case GTIN -> consulta.gtin();
            case PALABRAS -> unir(consulta.marca(), consulta.palabras());

        };
        if (texto == null) {
            return null;
        }
        texto = texto.replaceAll("\\s+", " ").trim();
        if (texto.isEmpty()) {
            return null;
        }
        return texto.length() <= LARGO_CONSULTA ? texto : texto.substring(0, LARGO_CONSULTA).trim();
    }

    public ConsultaFuente consultar(FuenteBusquedaConfig fuente, ConsultaBusqueda consulta) {
        CodigoFuente codigo = fuente.getCodigo();
        String nombre = codigo.name();
        if (!sesiones.configuracionCompleta(codigo)) {
            return ConsultaFuente.de(nombre, EstadoConsultaFuente.OMITIDA_SESION,
                    "Fuente omitida: sin configurar.", List.of());
        }
        int limite = limiteDe(fuente);
        ControlCuota cuota = cuotaDe(codigo);
        if (cuota.usadasHoy() >= limite) {
            return ConsultaFuente.de(nombre, EstadoConsultaFuente.OMITIDA_CUOTA,
                    fuente.getNombre() + " se omitiÃ³: sin cuota diaria.", List.of());
        }

        pausa.esperar(fuente.getPausaSegundos() == null ? 0 : fuente.getPausaSegundos());

        Optional<String> sesion = pedirSesion(codigo);
        if (sesion.isEmpty()) {
            avisarSesionNoRenovada(codigo);

                    fuente.getNombre() + " se omitiÃ³: sin sesiÃ³n vigente.", List.of());
        }

        List<SelectorFuenteDto> selectores = selectoresDe(codigo);
        ResultadoLectura lectura = invocarBuscar(codigo, fuente.getUrlBase(), sesion.get(), consulta, selectores);

            return ConsultaFuente.de(nombre, EstadoConsultaFuente.NO_CONECTADO,
                    lectura.mensaje() != null ? lectura.mensaje() : LectorFuenteNoConectado.MENSAJE, List.of());
        }

        try {
            cuota.reservar(limite);
        } catch (FuenteNoDisponibleException e) {

                    fuente.getNombre() + " se omitiÃ³: sin cuota diaria.", List.of());
        }

        if (lectura.tipo() == ResultadoLectura.Tipo.ERROR && lectura.reintentable()) {
            lectura = invocarBuscar(codigo, fuente.getUrlBase(), sesion.get(), consulta, selectores);
        }
        return aplicar(fuente, consulta, lectura);
    }

    private ConsultaFuente aplicar(FuenteBusquedaConfig fuente, ConsultaBusqueda consulta,
                                   ResultadoLectura lectura) {
        String nombre = fuente.getCodigo().name();
        return switch (lectura.tipo()) {
            case NO_CONECTADO -> ConsultaFuente.de(nombre, EstadoConsultaFuente.NO_CONECTADO,
                    lectura.mensaje() != null ? lectura.mensaje() : LectorFuenteNoConectado.MENSAJE, List.of());
            case BLOQUEADA -> ConsultaFuente.de(nombre, EstadoConsultaFuente.BLOQUEADA,
                    lectura.mensaje() != null ? lectura.mensaje() : "La tienda bloqueÃ³ la consulta.", List.of());
            case VERIFICACION -> {
                String motivo = lectura.mensaje() != null ? lectura.mensaje()
                        : "La tienda pidiÃ³ un cÃ³digo o CAPTCHA al buscar.";
                sesiones.marcarVerificacion(fuente.getCodigo(), motivo);
                avisos.avisarSiCorresponde(nombre, EstadoFuente.REQUIERE_VERIFICACION.name(), motivo);

                        lectura.mensaje(), List.of());
